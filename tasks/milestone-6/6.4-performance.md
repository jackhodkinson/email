### 3. Virtual List for Large Inboxes

```typescript
// Use react-window or similar for large lists
import { FixedSizeList } from "react-window";

function VirtualEmailList({ emails }) {
  return (
    <FixedSizeList
      height={600}
      itemCount={emails.length}
      itemSize={80}
    >
      {({ index, style }) => (
        <EmailItem email={emails[index]} style={style} />
      )}
    </FixedSizeList>
  );
}
```
