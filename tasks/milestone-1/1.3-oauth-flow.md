### 2. OAuth Auth Module (`src/lib/gmail/auth.ts`)

Create OAuth helper functions:

```typescript
import { google } from "googleapis";

// Create OAuth2 client
export function createOAuth2Client() {
  return new google.auth.OAuth2(
    process.env.GOOGLE_CLIENT_ID,
    process.env.GOOGLE_CLIENT_SECRET,
    process.env.GOOGLE_REDIRECT_URI
  );
}

// Generate auth URL for consent screen
export function getAuthUrl(): string {
  const oauth2Client = createOAuth2Client();
  return oauth2Client.generateAuthUrl({
    access_type: "offline", // Get refresh token
    scope: ["https://www.googleapis.com/auth/gmail.readonly"],
    prompt: "consent", // Always show consent (ensures refresh token)
  });
}

// Exchange auth code for tokens
export async function exchangeCodeForTokens(code: string) {
  const oauth2Client = createOAuth2Client();
  const { tokens } = await oauth2Client.getToken(code);
  return tokens;
}

// Get authenticated client for an account
export async function getAuthenticatedClient(account: Account) {
  const oauth2Client = createOAuth2Client();
  oauth2Client.setCredentials({
    access_token: account.access_token,
    refresh_token: account.refresh_token,
    expiry_date: account.token_expiry * 1000, // Convert to ms
  });

  // Handle token refresh
  oauth2Client.on("tokens", async (tokens) => {
    // Update tokens in database
    await updateAccountTokens(account.id, tokens);
  });

  return oauth2Client;
}
```

### 3. Server Functions (`src/server/functions.ts`)

Create TanStack server functions for OAuth:

```typescript
import { createServerFn } from "@tanstack/react-start/server";

// Get the Google OAuth URL
export const getGoogleAuthUrl = createServerFn({ method: "GET" })
  .handler(async () => {
    return { url: getAuthUrl() };
  });

// Handle OAuth callback (exchange code for tokens)
export const handleOAuthCallback = createServerFn({ method: "GET" })
  .validator((data: { code: string }) => data)
  .handler(async ({ data }) => {
    const tokens = await exchangeCodeForTokens(data.code);

    // Get user's email address
    const oauth2Client = createOAuth2Client();
    oauth2Client.setCredentials(tokens);
    const gmail = google.gmail({ version: "v1", auth: oauth2Client });
    const profile = await gmail.users.getProfile({ userId: "me" });

    // Store account in database
    const accountId = crypto.randomUUID();
    await upsertAccount({
      id: accountId,
      email: profile.data.emailAddress!,
      access_token: tokens.access_token!,
      refresh_token: tokens.refresh_token!,
      token_expiry: Math.floor(tokens.expiry_date! / 1000),
    });

    return { success: true, accountId, email: profile.data.emailAddress };
  });
```

### 4. OAuth Callback Route (`src/routes/auth/callback.tsx`)

Handle the redirect from Google:

```typescript
import { createFileRoute, redirect } from "@tanstack/react-router";
import { handleOAuthCallback } from "../../server/functions";

export const Route = createFileRoute("/auth/callback")({
  validateSearch: (search: Record<string, unknown>) => ({
    code: search.code as string,
    error: search.error as string | undefined,
  }),
  loaderDeps: ({ search }) => ({ code: search.code, error: search.error }),
  loader: async ({ deps }) => {
    if (deps.error) {
      throw redirect({ to: "/", search: { error: deps.error } });
    }

    if (!deps.code) {
      throw redirect({ to: "/", search: { error: "missing_code" } });
    }

    try {
      const result = await handleOAuthCallback({ data: { code: deps.code } });
      // Redirect to inbox after successful auth
      throw redirect({ to: "/" });
    } catch (e) {
      throw redirect({ to: "/", search: { error: "auth_failed" } });
    }
  },
  component: () => null, // Never renders, always redirects
});
```

## Technical Notes

### Refresh Tokens

- Only returned on first consent or when using `prompt: "consent"`
- Store securely - it's long-lived
- Use it to get new access tokens when they expire

### Token Expiry

- Access tokens expire in ~1 hour
- The `googleapis` library auto-refreshes when using `setCredentials`
- Listen to `tokens` event to persist new tokens

### Error Handling

Google may return these errors:
- `access_denied` - User clicked "Cancel"
- `invalid_grant` - Code already used or expired

### Server-Side Only

OAuth token exchange MUST happen server-side. Never expose:
- Client secret
- Refresh tokens
- Access tokens

## Troubleshooting

### "redirect_uri_mismatch"
- Check `.env.local` `GOOGLE_REDIRECT_URI` matches Google Cloud Console exactly

### No refresh token
- Delete app access at https://myaccount.google.com/permissions
- Re-auth with `prompt: "consent"`

### "Token has been expired or revoked"
- Refresh token may be invalid
- Re-authenticate the account
