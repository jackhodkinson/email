## Goal

Create a global keyboard event handler for vim-style navigation.

## Context

- Keyboard shortcuts defined in [MVP.md](../../MVP.md#keyboard-navigation)
- Should work across the entire app
- Must not interfere with text inputs

## Requirements

### 1. Keyboard Hook (`src/lib/hooks/use-keyboard.ts`)

```typescript
export function useKeyboard(handlers: KeyboardHandlers) {
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      // Ignore if in input/textarea
      if (isInputFocused()) return;

      // Handle key
      const handler = handlers[e.key];
      if (handler) {
        e.preventDefault();
        handler();
      }
    };

    window.addEventListener("keydown", handleKeyDown);
    return () => window.removeEventListener("keydown", handleKeyDown);
  }, [handlers]);
}
```
